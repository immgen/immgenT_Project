MyFeaturePlotHighlight <- function (seurat_object, feature, cells_to_highlight, umap_to_plot = "umap", 
    assay = "RNA", colors = viridis::viridis(10), title = "", 
    pixels = c(512, 512), highlight_size = 1, highlight_alpha = 1, 
    xlim = NULL, ylim = NULL, background_alpha = 0.1) 
{
    dim_data <- seurat_object[[umap_to_plot]]@cell.embeddings
    dim1 <- dim_data[, 1]
    dim2 <- dim_data[, 2]
    tmp_all <- data.frame(seurat_object@meta.data, dim1 = dim1, 
        dim2 = dim2)
    p_base <- ggplot2::ggplot(tmp_all) + scattermore::geom_scattermore(ggplot2::aes(x = dim1, 
        y = dim2), color = "grey", pointsize = 1, alpha = background_alpha, 
        pixels = pixels)
    feature_expr <- Seurat::GetAssayData(seurat_object, assay = assay, 
        layer = "data")[feature, ]
    tmp_highlight <- tmp_all[cells_to_highlight, ]
    tmp_highlight$feature_value <- feature_expr[cells_to_highlight]
    p_final <- p_base + ggplot2::geom_point(data = tmp_highlight, 
        ggplot2::aes(x = dim1, y = dim2, color = feature_value), 
        size = highlight_size, alpha = highlight_alpha) + ggplot2::scale_color_gradientn(colours = colors, 
        name = feature) + ggplot2::theme_void() + NoGrid() + 
        ggplot2::labs(title = title, x = paste0(umap_to_plot, 
            "_1"), y = paste0(umap_to_plot, "_2"))
    if (!is.null(xlim) || !is.null(ylim)) {
        p_final <- p_final + ggplot2::coord_cartesian(xlim = xlim, 
            ylim = ylim)
    }
    return(p_final)
}

# run_kappa_heatmap.R
#
# Standalone script: plots the kappa joint-tail heatmap for all pairwise
# TH program comparisons.
#
#   kappa(x, y) = P(X > x, Y > y)          <- numerator:   empirical joint tail
#                 ──────────────────
#                 P(X > x) * P(Y > y)       <- denominator: independence baseline
#
#   kappa < 1  =>  joint tail DEPLETION   (programs mutually exclusive at extremes)
#   kappa = 1  =>  independence
#   kappa > 1  =>  joint tail ENRICHMENT  (programs co-occur at extremes)
#
# Output: one PDF per pair in figure/kappa_heatmap/

source("code/tail_dependence_functions.R")

short_label <- function(nm) {
  sub("_Score1$", "", nm)
}

pretty_module_label <- function(score_name) {
  lab <- short_label(score_name)

  if (toupper(lab) == "TH17") {
    return("Th17-like module score")
  }

  if (toupper(lab) == "TH1") {
    return("Th1-like module score")
  }

  if (toupper(lab) == "TH2") {
    return("Th2-like module score")
  }

  if (toupper(lab) == "TFH_TCHRON") {
    return("Tfh/Tchron-like module score")
  }

  sprintf("%s-like module score", lab)
}

MyFeaturePlotHighlight <- function (seurat_object, feature, cells_to_highlight, umap_to_plot = "umap", 
    assay = "RNA", colors = viridis::viridis(10), title = "", 
    pixels = c(512, 512), highlight_size = 1, highlight_alpha = 1, 
    xlim = NULL, ylim = NULL, background_alpha = 0.1) 
{
    dim_data <- seurat_object[[umap_to_plot]]@cell.embeddings
    dim1 <- dim_data[, 1]
    dim2 <- dim_data[, 2]
    tmp_all <- data.frame(seurat_object@meta.data, dim1 = dim1, 
        dim2 = dim2)
    p_base <- ggplot2::ggplot(tmp_all) + scattermore::geom_scattermore(ggplot2::aes(x = dim1, 
        y = dim2), color = "grey", pointsize = 1, alpha = background_alpha, 
        pixels = pixels)
    feature_expr <- Seurat::GetAssayData(seurat_object, assay = assay, 
        layer = "data")[feature, ]
    tmp_highlight <- tmp_all[cells_to_highlight, ]
    tmp_highlight$feature_value <- feature_expr[cells_to_highlight]
    p_final <- p_base + ggplot2::geom_point(data = tmp_highlight, 
        ggplot2::aes(x = dim1, y = dim2, color = feature_value), 
        size = highlight_size, alpha = highlight_alpha) + ggplot2::scale_color_gradientn(colours = colors, 
        name = feature) + ggplot2::theme_void() + NoGrid() + 
        ggplot2::labs(title = title, x = paste0(umap_to_plot, 
            "_1"), y = paste0(umap_to_plot, "_2"))
    if (!is.null(xlim) || !is.null(ylim)) {
        p_final <- p_final + ggplot2::coord_cartesian(xlim = xlim, 
            ylim = ylim)
    }
    return(p_final)
}
